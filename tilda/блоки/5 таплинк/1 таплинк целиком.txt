<div id="taplink-root" style="min-height:100vh;background:#FFFFFF"></div>
<script>(function(){var B='https://cdn.jsdelivr.net/gh/kupdasha/my-site@main/',v=Math.floor(Date.now()/36e5),root=document.getElementById('taplink-root');
try{if(localStorage.getItem('tl-mode')==='fun')root.style.background='#0E0F12'}catch(e){}
fetch(B+'taplink.html?v='+v).then(function(r){return r.text()}).then(function(html){
var doc=new DOMParser().parseFromString(html,'text/html');
[].slice.call(doc.querySelectorAll('head link, head style, body > link, body > style')).forEach(function(n){document.head.appendChild(n)});
[].slice.call(doc.body.childNodes).forEach(function(n){root.appendChild(n)});
[].slice.call(root.querySelectorAll('script')).forEach(function(s){var x=document.createElement('script');x.textContent=s.textContent;s.parentNode.replaceChild(x,s)});
root.style.minHeight='';root.style.background=''})})()</script>
